# C-program-to-perform-basic-arithmetic-operations
C program to perform basic arithmetic operations

// Simple C program to perform basic arithmetic operations

#include <stdio.h>
int main()
{
    int a, b;
    printf("Enter the first number a:\n", a);
    scanf("%d", &a);
    printf("Enter the second number b:\n", b);
    scanf("%d", &b);

    printf("a+b=%d\n", a + b);
    printf("a-b=%d\n", a - b);
    printf("a*b=%d\n", a * b);
    printf("a/b=%d\n", a / b);

    return 0;
}
